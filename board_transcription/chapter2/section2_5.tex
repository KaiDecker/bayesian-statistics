\boardsection{5}{预测}
\block{先验预测与后验预测}
\[
m(z)=\int g(z\mid\theta)\pi(\theta)\,d\theta,\qquad
m(z\mid x)=\int g(z\mid\theta)\pi(\theta\mid x)\,d\theta.
\]
\[
m(z\mid x)=\E[g(z\mid\theta)\mid x]
\ne g(z\mid\E(\theta\mid x)).
\]
\[
\E(Z\mid x)=\E[\E(Z\mid\theta)\mid x],
\]
\[
\Var(Z\mid x)=
\E[\Var(Z\mid\theta)\mid x]+\Var[\E(Z\mid\theta)\mid x].
\]
三个分布：
\[
\underbrace{\pi(\theta\mid x)}_{\text{参数的后验分布}},\qquad
g(z\mid\theta),\qquad
\underbrace{m(z\mid x)}_{\text{未来观测的后验预测}}.
\]

\block{正态预测}
\[
Z\mid\theta\sim N(\theta,\sigma^2),\qquad
\theta\mid x\sim N(\mu_n,\tau_n^2).
\]
\[
\E(Z\mid x)=\E(\theta\mid x)=\mu_n,
\]
\[
\Var(Z\mid x)=\E(\sigma^2\mid x)+\Var(\theta\mid x)
=\sigma^2+\tau_n^2.
\]
\[
Z\mid x\sim N(\mu_n,\sigma^2+\tau_n^2).
\]
参数$\theta$的95\%后验可信区间：
\[
\mu_n\pm1.96\tau_n.
\]
未来观测$Z$的95\%后验预测区间：
\[
\mu_n\pm1.96\sqrt{\sigma^2+\tau_n^2}.
\]

在平方损失下，最优点预测为预测均值
\[
\hat Z=\E(Z\mid x).
\]
在绝对损失下，最优点预测为预测中位数。

\block{例2.5.1}
$\theta$：篮球胜的概率。$X$：过去$n$次中的获胜次数（观测值$x$）。
$Z$：未来$k$次中的获胜次数（未观测）。
\[
X\mid\theta\sim\Bin(n,\theta),\qquad
Z\mid\theta\sim\Bin(k,\theta).
\]
\[
n=10,\quad x=3,\quad k=5.
\]
取共轭先验$\theta\sim\Be(\alpha,\beta)$，
\[
\pi(\theta\mid x)\propto
\theta^{x+\alpha-1}(1-\theta)^{n-x+\beta-1},
\quad \theta\mid x\sim\Be(a,b),
\quad a=x+\alpha,\ b=n-x+\beta.
\]
取$\alpha=\beta=1$（均匀先验），
\[
\theta\mid X=3\sim\Be(4,8).
\]
\begin{align*}
P(Z=z\mid x)
&=\int_0^1P(Z=z\mid\theta)\pi(\theta\mid x)\,d\theta\\
&=\frac{\binom kz}{B(a,b)}
\int_0^1\theta^{a+z-1}(1-\theta)^{b+k-z-1}\,d\theta\\
&=\binom kz\frac{B(a+z,b+k-z)}{B(a,b)},\qquad z=0,\ldots,k\\
&=\binom kz
\frac{\Gamma(a+b)\Gamma(a+z)\Gamma(b+k-z)}
{\Gamma(a)\Gamma(b)\Gamma(a+b+k)}.
\end{align*}
代入$a=4,b=8,k=5$，
\[
P(Z=z\mid X=3)=\binom5z
\frac{\Gamma(12)\Gamma(4+z)\Gamma(13-z)}
{\Gamma(4)\Gamma(8)\Gamma(17)}.
\]
\begin{align*}
P(Z=0\mid X=3)
&=\frac{\Gamma(12)\Gamma(13)}{\Gamma(8)\Gamma(17)}
=\frac{11!\,12!}{7!\,16!}
=\frac{33}{182}\approx0.1813,\\
P(Z=1\mid X=3)&\approx0.3022.
\end{align*}
①猜中具体次数：点预测$1$次。

②采用平方损失，使用预测均值：
\[
\E(Z\mid x)=\E[\E(Z\mid\theta)\mid x]=k\E(\theta\mid x).
\]
\[
\E(Z\mid X=3)=5\times\frac4{12}=\frac53.
\]
另外，
\[
P(0\leq Z\leq3\mid X=3)
=\frac{99+165+154+90}{546}
=\frac{12}{13}\approx0.9231.
\]
